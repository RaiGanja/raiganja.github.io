> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Вместо редиректа на `/wow/` в корне gabella.site — витрина гидов на ru/en/uk со своим SEO.

**Architecture:** Маленький сборщик на Python: `content.py` (данные и тексты) + `index.src.html` (шаблон с `@@КЛЮЧ@@`) → `build.py` пишет `index.html`, `en/index.html`, `uk/index.html`, `sitemap.xml`, `robots.txt`. Шрифты, арт, иконки и `news.json` берутся из гида по абсолютным путям `/wow/assets/…`, в хаб не копируются.

**Tech Stack:** Python 3.14 (только stdlib + Pillow 13 для OG), `unittest` (pytest не установлен), headless Chrome (`C:\Program Files\Google\Chrome\Application\chrome.exe`), puppeteer-core и Lighthouse 12.8.2 в `%TEMP%\wflh`.

- [ ] **Step 5: Написать `build.py`**

```python
"""Сборка хаба gabella.site: python build.py"""
import html as H
import os
import sys

import content as C

ROOT = os.path.dirname(os.path.abspath(__file__))
OUT = {'ru': 'index.html', 'en': 'en/index.html', 'uk': 'uk/index.html'}
CF_BEACON_TOKEN = ''


def url(lang):
    return f'{C.SITE}/{C.PREFIX[lang]}'


def guide(lang):
    return C.GAMES[0]['path'] + C.PREFIX[lang]


def esc(s):
    return H.escape(s, quote=True)


def hreflang():
    links = [f'<link rel="alternate" hreflang="{l}" href="{url(l)}">' for l in C.LANGS]
    links.append(f'<link rel="alternate" hreflang="x-default" href="{C.SITE}/">')
    return '\n'.join(links)


def body(lang):
    return f'<h1>{esc(C.T[lang]["h1"])}</h1>'


def css():
    return ''


def render(lang):
    if len(C.GAMES) != 1:
        sys.exit('build.py: в GAMES больше одной игры — сначала нужна сетка игр (спека, «Следующая игра»)')
    t = C.T[lang]
    with open(os.path.join(ROOT, 'index.src.html'), encoding='utf-8') as f:
        page = f.read()
    alt_loc = '\n'.join(f'<meta property="og:locale:alternate" content="{C.LOCALE[l]}">' for l in C.LANGS if l != lang)
    subs = {
        '/*@@CSS@@*/': css(),
        '@@BODY@@': body(lang),
        '@@HREFLANG@@': hreflang(),
        '@@OG_ALTLOC@@': alt_loc,
        '@@LANG@@': lang,
        '@@TITLE@@': esc(t['title']),
        '@@DESC@@': esc(t['desc']),
        '@@URL@@': url(lang),
        '@@OG@@': f'{C.SITE}/assets/og-{lang}.jpg',
        '@@OG_ALT@@': esc(t['og_alt']),
        '@@LOCALE@@': C.LOCALE[lang],
    }
    for k, v in subs.items():
        page = page.replace(k, v)
    if '@@' in page:
        sys.exit(f'build.py: в странице {lang} остались плейсхолдеры @@')
    return page


def write(rel, text):
    path = os.path.join(ROOT, rel)
    os.makedirs(os.path.dirname(path) or ROOT, exist_ok=True)
    with open(path, 'w', encoding='utf-8', newline='\n') as f:
        f.write(text)


def main():
    for lang in C.LANGS:
        write(OUT[lang], render(lang))
    print('hub: собрано', ', '.join(OUT.values()))


if __name__ == '__main__':
    main()
```

Важно: в `subs` CSS и BODY подставляются первыми, а остальные ключи — после них. Так `@@LANG@@` и `@@URL@@` внутри тела тоже заменятся. Значения текстов не должны содержать `@@`.

import json


def json_ld(lang):
    g = C.GAMES[0]
    data = {
        '@context': 'https://schema.org', '@type': 'WebSite', 'name': 'Gabella',
        'url': url(lang), 'inLanguage': lang, 'description': C.T[lang]['desc'],
        'hasPart': [{'@type': 'WebSite', 'name': g['name'][lang], 'url': C.SITE + guide(lang)}],
    }
    return '<script type="application/ld+json">' + json.dumps(data, ensure_ascii=False) + '</script>'


def beacon():
    if not CF_BEACON_TOKEN:
        return ''
    b = json.dumps({'token': CF_BEACON_TOKEN})
    return f'<script type="module" src="https://static.cloudflareinsights.com/beacon.min.js" data-cf-beacon=\'{b}\'></script>'


def sitemap():
    alts = ''.join(f'<xhtml:link rel="alternate" hreflang="{l}" href="{url(l)}"/>' for l in C.LANGS)
    alts += f'<xhtml:link rel="alternate" hreflang="x-default" href="{C.SITE}/"/>'
    urls = ''.join(f'<url><loc>{url(l)}</loc>{alts}</url>\n' for l in C.LANGS)
    return ('<?xml version="1.0" encoding="UTF-8"?>\n'
            '<urlset xmlns="http://www.sitemaps.org/schemas/sitemap/0.9" xmlns:xhtml="http://www.w3.org/1999/xhtml">\n'
            + urls + '</urlset>\n')


def robots():
    return f'User-agent: *\nAllow: /\n\nSitemap: {C.SITE}/sitemap.xml\nSitemap: {C.SITE}/wow/sitemap.xml\n'

```python
"""Проверка собранного хаба: python tests/check_site.py [--live]
--live: каждая ссылка и ассет /wow/... со страниц хаба отвечает 200 на https://gabella.site"""
import os
import re
import sys
import urllib.request
import xml.etree.ElementTree as ET

ROOT = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
sys.path.insert(0, ROOT)
import build  # noqa: E402
import content as C  # noqa: E402

fails = []


def check(ok, msg):
    if not ok:
        fails.append(msg)


def read(rel):
    with open(os.path.join(ROOT, rel), encoding='utf-8') as f:
        return f.read()


def wow_refs(html):
    refs = set(re.findall(r'''(?:href|src)="(/wow/[^"]*)"''', html))
    refs |= set(re.findall(r'url\((/wow/[^)]*)\)', html))
    for srcset in re.findall(r'srcset="([^"]*)"', html):
        refs |= {p.split()[0] for p in srcset.split(',') if p.strip().startswith('/wow/')}
    refs |= set(re.findall(r"fetch\('(/wow/[^']*)'\)", html))
    return refs


def main():
    live = '--live' in sys.argv
    refs = set()
    for lang in C.LANGS:
        html = read(build.OUT[lang])
        check(html == build.render(lang), f'{lang}: собранный файл устарел — запусти python build.py')
        check('@@' not in html, f'{lang}: остались @@')
        check(f'<link rel="canonical" href="{build.url(lang)}">' in html, f'{lang}: canonical')
        for other in C.LANGS + ['x-default']:
            check(f'hreflang="{other}"' in html, f'{lang}: нет hreflang {other}')
        check(os.path.exists(os.path.join(ROOT, 'assets', f'og-{lang}.jpg')), f'{lang}: нет og-{lang}.jpg')
        refs |= wow_refs(html)
    ns = {'s': 'http://www.sitemaps.org/schemas/sitemap/0.9'}
    check(len(ET.fromstring(read('sitemap.xml')).findall('s:url', ns)) == 3, 'sitemap: не 3 URL')
    check(read('robots.txt').count('Sitemap:') == 2, 'robots: не 2 строки Sitemap')
    check(read('CNAME').strip() == 'gabella.site', 'CNAME')
    if live:
        for ref in sorted(refs):
            try:
                code = urllib.request.urlopen(urllib.request.Request(C.SITE + ref, method='HEAD'), timeout=20).status
            except urllib.error.HTTPError as e:
                code = e.code
            check(code == 200, f'live {ref}: {code}')
        print(f'live: проверено {len(refs)} адресов /wow/')
    if fails:
        print('\n'.join('FAIL ' + f for f in fails))
        sys.exit(1)
    print('check_site: OK')


if __name__ == '__main__':
    main()
```
